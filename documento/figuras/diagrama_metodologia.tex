% Miniatura: \mini[ancho]{archivo}{escala de color: v (viridis), g (grises) o n (ninguna)}
% Sin elementos superpuestos a la imagen; si corresponde, la escala de color va al lado.
\newcommand{\mini}[3][1.3cm]{%
\begin{tikzpicture}[baseline=(i.center), every node/.style={}, sharp corners]
  \node[inner sep=0, anchor=south west] (i) at (0,0) {\includegraphics[width=#1]{#2}};
  \ifx v#3
    \foreach \k in {0,...,63}{%
      \pgfmathtruncatemacro{\tramo}{\k/16}\pgfmathsetmacro{\frac}{100*(\k-16*\tramo)/16}%
      \ifcase\tramo \colorlet{cb}{v2!\frac!v1}\or \colorlet{cb}{v3!\frac!v2}\or \colorlet{cb}{v4!\frac!v3}\else \colorlet{cb}{v5!\frac!v4}\fi
      \fill[cb, draw=none] ($(i.south east)+(0.07,\k*#1/64)$) rectangle ++(0.12,#1/64+0.01pt);}
    \draw[line width=0.2pt] ($(i.south east)+(0.07,0)$) rectangle ++(0.12,#1);
  \fi
  \ifx g#3
    \foreach \k in {0,...,63}{\pgfmathsetmacro{\frac}{100*\k/63}\colorlet{cb}{white!\frac!black}%
      \fill[cb, draw=none] ($(i.south east)+(0.07,\k*#1/64)$) rectangle ++(0.12,#1/64+0.01pt);}
    \draw[line width=0.2pt] ($(i.south east)+(0.07,0)$) rectangle ++(0.12,#1);
  \fi
\end{tikzpicture}}
